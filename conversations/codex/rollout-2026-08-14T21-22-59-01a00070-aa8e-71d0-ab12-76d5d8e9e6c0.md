        by_bar.setdefault(bar_idx, []).append(tr)
    out: list[Bar] = []

        ts = by_bar[bar_idx]
        prices = [t.price_ticks for t in ts]
        out.append(
            Bar(
                open=prices[0],
                high=max(prices),
                low=min(prices),
                close=prices[-1],
                volume=sum(t.quantity_units for t in ts),
                trade_count=len(ts),
